## Explanation

For every word, we have to verify whether we can form it by taking
at most one letter from each of the given blocks. Since we are
unaware of which block to choose first, we will enumerate over
their permutations, as there are only $4! = 24$ permutations of blocks. For each
permutation, we will greedily search through the blocks in that
order: whenever we get a match, we search for the next letter;
otherwise, we skip it.

## Implementation

**Time Complexity:** $\mathcal{O}(N)$

```java

import java.io.*;
import java.util.*;

public class Main {
    // cnt[i][j] = number of times letter j appears on block i
    static int[][] cnt = new int[4][26];

    static boolean isPossible(String word) {
        // Try every possible ordering of the 4 blocks
        int[] perm = {0, 1, 2, 3};

        do {
            int j = 0;

            // Try to form the word using this block ordering
            for (int i = 0; i < 4; i++) {
                if (cnt[perm[i]][word.charAt(j) - 'A'] > 0) {
                    j++;
                }

                // All characters of the word were matched
                if (j == word.length()) {
                    return true;
                }
            }
        } while (nextPermutation(perm));

        // No ordering of the blocks can form the word
        return false;
    }

    // Generates the next lexicographically greater permutation
    static boolean nextPermutation(int[] a) {
        int i = a.length - 2;

        // Find the first position from the right where a[i] < a[i + 1]
        while (i >= 0 && a[i] >= a[i + 1]) {
            i--;
        }

        // No more permutations
        if (i < 0) {
            return false;
        }

        // Find the smallest element greater than a[i]
        int j = a.length - 1;
        while (a[j] <= a[i]) {
            j--;
        }

        // Swap a[i] and a[j]
        int temp = a[i];
        a[i] = a[j];
        a[j] = temp;

        // Reverse the suffix to get the next permutation
        for (int l = i + 1, r = a.length - 1; l < r; l++, r--) {
            temp = a[l];
            a[l] = a[r];
            a[r] = temp;
        }

        return true;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());

        // Read the 4 blocks and count the letters on each block
        for (int i = 0; i < 4; i++) {
            String block = br.readLine();

            for (char c : block.toCharArray()) {
                cnt[i][c - 'A']++;
            }
        }

        // Check each word
        for (int i = 0; i < n; i++) {
            String word = br.readLine();

            System.out.println(isPossible(word) ? "YES" : "NO");
        }
    }
}
```

```py

from itertools import permutations

# cnt[i][j] = number of times letter j appears on block i
cnt = [[0] * 26 for _ in range(4)]

n = int(input())

# Read the 4 blocks and count the letters on each block
for i in range(4):
    block = input()

    for c in block:
        cnt[i][ord(c) - ord('A')] += 1


def is_possible(word):
    # Try every possible ordering of the 4 blocks
    for perm in permutations(range(4)):
        j = 0

        # Try to form the word using this block ordering
        for i in range(4):
            if cnt[perm[i]][ord(word[j]) - ord('A')]:
                j += 1

            # All characters of the word were matched
            if j == len(word):
                return True

    # No ordering of the blocks can form the word
    return False


# Check each word
for _ in range(n):
    word = input()
    print("YES" if is_possible(word) else "NO")
```
